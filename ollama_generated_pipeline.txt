**SCRIPT 1: fetch_1year_m1_data.py**

**SCRIPT 2: backtest_1year_all41.py**
```python
import pandas as pd

# Load all parquet files
data_files = ['data_1y_EURUSD_M1.parquet', 'data_1y_GBPUSD_M1.parquet', ...]
dfs = []
for file in data_files:
    df = pd.read_parquet(file)
    dfs.append(df)

# Concatenate all dataframes
df_all = pd.concat(dfs, ignore_index=True)

# Run ALL 41 strategies (same logic as backtest_1week_all41.py)
strategies = ['ORDER_BLOCK_REVERSAL', 'DAY_LOW_BULLISH', ...]
for strategy in strategies:
    # Compute real ATR from rolling 14-bar std of close
    atr_mult = 0.5

    # For each bar: compute SL and TP points
    for i, row in df_all.iterrows():
        entry = row['close']
        sl_atr_mult = atr_mult * row['atr']
        tp_atr_mult = atr_mult * row['atr']

        # Compute SL and TP points
        sl = entry - sl_atr_mult if row['direction'] == 'BUY' else entry + sl_atr_mult
        tp = entry + tp_atr_mult if row['direction'] == 'BUY' else entry - tp_atr_mult

        # Add features to each signal row: hour, weekday, session (ASIAN/LONDON/NY/US), rsi_at_signal, adx_at_signal, atr_value
        df_all.loc[i, 'hour'] = row['time'].hour
        df_all.loc[i, 'weekday'] = row['time'].weekday()
        df_all.loc[i, 'session'] = 'ASIAN' if 0 <= i < 240 else 'LONDON' if 240 <= i < 480 else 'NY' if 480 <= i < 720 else 'US'
        df_all.loc[i, 'rsi_at_signal'] = row['rsi']
        df_all.loc[i, 'adx_at_signal'] = row['adx']
        df_all.loc[i, 'atr_value'] = row['atr']

    # Save results to: backtest_1year_signals.csv
df_all.to_csv('backtest_1year_signals.csv', index=False)
```

**SCRIPT 3: ml_walkforward_trainer.py**
```python
import pandas as pd
from sklearn.ensemble import GradientBoostingClassifier, RandomForestClassifier, XGBClassifier
from sklearn.model_selection import GridSearchCV
from joblib import dump

# Load backtest_1year_signals.csv
df = pd.read_csv('backtest_1year_signals.csv')

# Walk-forward split: train on first 270 days, validate on last 95 days
train_df = df.iloc[:270]
val_df = df.iloc[270:]

# Define features and target
features = ['symbol', 'strategy', 'direction', 'hour', 'weekday', 'session', 'rsi_at_signal', 'adx_at_signal', 'atr_value', 'sl_pts', 'tp_pts']
target = 'outcome'

# Train 3 models: GradientBoostingClassifier, RandomForestClassifier, XGBClassifier
models = [GradientBoostingClassifier(), RandomForestClassifier(), XGBClassifier()]
param_grid = {'n_estimators': [10, 50, 100], 'max_depth': [3, 5, 7]}

for model in models:
    grid_search = GridSearchCV(model, param_grid, cv=3, scoring='f1_weighted')
    grid_search.fit(train_df[features], train_df[target])
    print(f"Best {model.__class__.__name__} parameters: {grid_search.best_params_}")
    dump(grid_search.best_estimator_, 'ml_1year_best_model.joblib')

# Save feature importances to: ml_feature_importance.json
importances = []
for model in models:
    importance = grid_search.best_estimator_.feature_importances_
    importances.append(importance)
with open('ml_feature_importance.json', 'w') as f:
    json.dump(importances, f)

print("ML training complete")
```

**SCRIPT 4: report_1year_results.py**
```python
import pandas as pd

# Load backtest_1year_signals.csv and ml_1year_best_model.joblib
df = pd.read_csv('backtest_1year_signals.csv')
model = joblib.load('ml_1year_best_model.joblib')

# Apply ML lot sizing: if win_prob < 0.55: lot = 0.01 else lot = 0.02
lot_size = 0.01 if model.predict(df[features]) < 0.55 else 0.02

# Compute per-trade P&L using real ATR-based SL/TP points × lot × tick_value
pnl = (df['close'] - df['sl']) * lot_size + (df['tp'] - df['close']) * lot_size

# Tick value lookup from MT5 or use: EURUSD/GBPUSD/AUDUSD=10, USDJPY=9.1, GOLD=1, SILVER=50, BTCUSD=1, ETHUSD=0.1 per 0.01 lot
tick_values = {'EURUSD': 10, 'GBPUSD': 10, ...}

# Aggregate:
daily_roi_table = df.groupby('date')['pnl'].sum().reset_index()
strategy_breakdown = df.groupby('strategy')['trades', 'wins', 'P&L'].sum().reset_index()
symbol_breakdown = df.groupby('symbol')['trades', 'wins', 'P&L'].sum().reset_index()

# Write to: ml_1year_backtest_report.md
with open('ml_1year_backtest_report.md', 'w') as f:
    f.write(f"**Daily ROI Table**\n{daily_roi_table.to_markdown()}\n")
    f.write(f"**Strategy Breakdown**\n{strategy_breakdown.to_markdown()}\n")
    f.write(f"**Symbol Breakdown**\n{symbol_breakdown.to_markdown()}\n")

print("Report generated and saved to ml_1year_backtest_report.md")
```
